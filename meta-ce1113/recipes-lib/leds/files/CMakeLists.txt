cmake_minimum_required(VERSION 3.16)
project(aurabot_leds VERSION 1.0.0 LANGUAGES C)
find_path(GPIO_INCLUDE_DIR libgpio.h REQUIRED)
find_library(GPIO_LIBRARY gpio REQUIRED)
add_library(leds SHARED src/libleds.c)
set_target_properties(leds PROPERTIES VERSION 1.0.0 SOVERSION 1)
target_include_directories(leds PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include PRIVATE ${GPIO_INCLUDE_DIR})
target_link_libraries(leds PRIVATE ${GPIO_LIBRARY})
install(TARGETS leds LIBRARY DESTINATION lib)
install(FILES include/libleds.h DESTINATION include)
