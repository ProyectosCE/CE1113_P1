cmake_minimum_required(VERSION 3.16)
project(aurabot_sensors VERSION 1.0.0 LANGUAGES C)
find_path(GPIO_INCLUDE_DIR libgpio.h REQUIRED)
find_library(GPIO_LIBRARY gpio REQUIRED)
add_library(sensors SHARED src/libsensors.c)
set_target_properties(sensors PROPERTIES VERSION 1.0.0 SOVERSION 1)
target_include_directories(sensors PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include PRIVATE ${GPIO_INCLUDE_DIR})
target_link_libraries(sensors PRIVATE ${GPIO_LIBRARY})
install(TARGETS sensors LIBRARY DESTINATION lib)
install(FILES include/libsensors.h DESTINATION include)
